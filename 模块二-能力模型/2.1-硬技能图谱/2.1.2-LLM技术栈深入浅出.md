## 三、核心概念图解

### 3.1 LLM 应用开发 vs 传统开发

```
传统开发：规则 + 数据 → 输出
  if question == "年假": return "10天"

LLM 应用：提示 + 数据 + 模型 → 输出
  prompt + context + LLM(question) → answer
```

### 3.2 RAG 工作流

```
用户问题
   ↓
[1] Query 改写（可选）
   ↓
[2] Embedding（向量化）
   ↓
[3] 向量库检索（TopK）
   ↓
[4] 重排序（Rerank，可选）
   ↓
[5] 构造 Prompt
   ↓
[6] LLM 生成
   ↓
[7] 后处理（引用、过滤）
   ↓
返回答案 + 引用
```

### 3.3 Agent 工作流（ReAct）

```
用户任务
   ↓
[1] LLM 思考（Thought）
   ↓
[2] 选择工具（Action）
   ↓
[3] 执行工具（Observation）
   ↓
[4] 是否完成？
   ├─ 否 → 回到 [1]
   └─ 是 → 输出最终结果
```

### 3.4 Function Calling 流程

```
用户：帮我查北京天气
   ↓
LLM 分析 → 决定调用 get_weather(city="北京")
   ↓
执行函数 → 返回 {"temp": 25, "weather": "晴"}
   ↓
LLM 用结果生成自然语言回答
   ↓
"北京今天 25 度，晴朗。"
```

```python
# 02_few_shot.py - FDE 必会的 Few-shot Prompt
from openai import OpenAI

client = OpenAI()

prompt = """将用户问题分类到以下类别之一：技术支持、账单问题、功能咨询、其他。

示例 1：
用户：登录不上去了
类别：技术支持

示例 2：
用户：这个月的发票怎么还没开？
类别：账单问题

示例 3：
用户：你们支持哪些功能？
类别：功能咨询

示例 4：
用户：你好
类别：其他

现在请分类：
用户：{}"""

def classify(question: str) -> str:
    response = client.chat.completions.create(
        model="gpt-4o-mini",
        messages=[{"role": "user", "content": prompt.format(question)}],
        temperature=0,
    )
    return response.choices[0].message.content.strip()

# 使用
```

```python
import json
from openai import OpenAI
from pydantic import BaseModel

client = OpenAI()


class Sentiment(BaseModel):
    sentiment: str  # positive / negative / neutral
    score: float    # 0-1
    keywords: list[str]


def analyze_sentiment(text: str) -> Sentiment:
    """情感分析（结构化输出）"""
    response = client.chat.completions.create(
        model="gpt-4o",
        messages=[
            {
                "role": "system",
                "content": """分析文本情感，输出 JSON：
                {"sentiment": "positive/negative/neutral", "score": 0-1, "keywords": ["关键词"]}""",
            },
            {"role": "user", "content": text},
        ],
        response_format={"type": "json_object"},
    )
    return Sentiment(**json.loads(response.choices[0].message.content))


# 使用
```

### 示例 4：基础 RAG（手写版）

```python
# 04_basic_rag.py - 不依赖框架，从零写 RAG
import chromadb
from openai import OpenAI
from openai import embeddings

client = OpenAI()
chroma = chromadb.PersistentClient(path="./chroma_db")
collection = chroma.get_or_create_collection("docs")


def add_document(content: str, doc_id: str):
    """添加文档到向量库"""
    collection.add(documents=[content], ids=[doc_id])


def query(question: str, top_k: int = 3) -> dict:
    """RAG 查询"""
    # 1. 检索
    results = collection.query(query_texts=[question], n_results=top_k)

    if not results["documents"][0]:
        return {"answer": "没找到相关资料", "sources": []}

    # 2. 构造 Prompt
    context = "\n\n".join(results["documents"][0])
    prompt = f"""基于以下资料回答问题。如果资料中没有，请说"不知道"。

资料：
{context}

问题：{question}
回答："""

    # 3. 生成
    response = client.chat.completions.create(
        model="gpt-4o-mini",
        messages=[{"role": "user", "content": prompt}],
    )

    return {
        "answer": response.choices[0].message.content,
        "sources": results["documents"][0],
    }


# 使用
add_document("公司年假政策：员工每年 10 天年假", "policy_1")
add_document("报销流程：通过财务系统提交，3 个工作日到账", "policy_2")

result = query("年假怎么休？")
print(result["answer"])
```

```python
# 05_langchain_rag.py
from langchain_community.document_loaders import DirectoryLoader
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain_openai import OpenAIEmbeddings, ChatOpenAI
from langchain_community.vectorstores import Chroma
from langchain.chains import RetrievalQA
from langchain.prompts import PromptTemplate

# 1. 加载文档
loader = DirectoryLoader("./docs", glob="**/*.txt")
documents = loader.load()

# 2. 分块
text_splitter = RecursiveCharacterTextSplitter(
    chunk_size=500,
    chunk_overlap=50,
)
chunks = text_splitter.split_documents(documents)

# 3. Embedding + 向量库
embeddings = OpenAIEmbeddings(model="text-embedding-3-small")
vectorstore = Chroma.from_documents(chunks, embeddings)

# 4. LLM
llm = ChatOpenAI(model="gpt-4o-mini", temperature=0)

# 5. 自定义 Prompt
prompt_template = """基于以下参考资料回答问题。如果资料中没有答案，请说"我不知道"。

参考资料：
{context}

问题：{question}
回答："""

PROMPT = PromptTemplate(
    template=prompt_template,
    input_variables=["context", "question"],
)

# 6. RAG Chain
qa_chain = RetrievalQA.from_chain_type(
    llm=llm,
    chain_type="stuff",
    retriever=vectorstore.as_retriever(search_kwargs={"k": 3}),
    chain_type_kwargs={"prompt": PROMPT},
    return_source_documents=True,
)

# 7. 查询
result = qa_chain({"query": "公司年假政策？"})
print("答案：", result["result"])
print("来源：", [doc.metadata for doc in result["source_documents"]])
```

```python
# 08_basic_agent.py - FDE 必会的 Agent
from langchain_openai import ChatOpenAI
from langchain.agents import tool, create_react_agent, AgentExecutor
from langchain.prompts import PromptTemplate
import requests


@tool
def get_weather(city: str) -> str:
    """查询天气"""
    # 实际项目调用天气 API
    return f"{city}今天 25 度，晴朗"


@tool
def search_docs(query: str) -> str:
    """搜索内部文档"""
    # 实际项目调用 RAG
    return f"关于 {query} 的内部资料：xxx"


tools = [get_weather, search_docs]

# LLM
llm = ChatOpenAI(model="gpt-4o", temperature=0)

# ReAct Prompt
react_prompt = PromptTemplate.from_template("""
回答用户问题，可以使用以下工具：

{tools}

使用以下格式：

Question: 用户问题
Thought: 思考要做什么
Action: 工具名
Action Input: 工具输入

... (Thought/Action/Observation 可重复)

Final Answer: 最终答案

Question: {input}
Thought:{agent_scratchpad}
""")

# 创建 Agent

agent_executor = AgentExecutor(agent=agent, tools=tools, verbose=True)
```

### 示例 9：Function Calling（OpenAI 原生）

```python
# 09_function_calling.py
import json
from openai import OpenAI

client = OpenAI()

# 定义工具
tools = [
    {
        "type": "function",
        "function": {
            "name": "get_weather",
            "description": "查询指定城市的天气",
            "parameters": {
                "type": "object",
                "properties": {
                    "city": {"type": "string", "description": "城市名称"},
                },
                "required": ["city"],
            },
        },
    },
    {
        "type": "function",
        "function": {
            "name": "search_docs",
            "description": "搜索内部文档",
            "parameters": {
                "type": "object",
                "properties": {
                    "query": {"type": "string", "description": "搜索关键词"},
                },
                "required": ["query"],
            },
        },
    },
]


def get_weather(city: str) -> dict:
    """实际查询天气"""
    return {"city": city, "temp": 25, "weather": "晴"}


def search_docs(query: str) -> str:
    """实际搜索文档"""
    return f"关于 {query} 的资料：xxx"


def run_conversation(user_message: str) -> str:
    """多轮对话 + Function Calling"""
    messages = [{"role": "user", "content": user_message}]

    while True:
        response = client.chat.completions.create(
            model="gpt-4o",
            messages=messages,
            tools=tools,
            tool_choice="auto",
        )

        response_message = response.choices[0].message
        tool_calls = response_message.tool_calls

        # 没有工具调用，返回最终答案
        if not tool_calls:
            return response_message.content

        # 执行工具调用
        messages.append(response_message)

        for tool_call in tool_calls:
            function_name = tool_call.function.name
            function_args = json.loads(tool_call.function.arguments)

            # 执行函数
            if function_name == "get_weather":
                function_response = json.dumps(get_weather(**function_args))
            elif function_name == "search_docs":
                function_response = search_docs(**function_args)
            else:
                function_response = "未知函数"

            messages.append({
                "tool_call_id": tool_call.id,
                "role": "tool",
                "content": function_response,
            })


# 使用
result = run_conversation("北京今天天气怎么样？")
print(result)
```

### 示例 10：Multi-Agent（多智能体）

```python
# 10_multi_agent.py - 多 Agent 协作
from langchain_openai import ChatOpenAI
from langgraph.graph import StateGraph, END
from typing import TypedDict, Annotated
import operator

llm = ChatOpenAI(model="gpt-4o", temperature=0)


# 状态
class AgentState(TypedDict):
    messages: Annotated[list, operator.add]
    next_agent: str


# 3 个 Agent
def research_agent(state: AgentState):
    """研究员 Agent"""
    response = llm.invoke(
        f"作为研究员，调研这个问题：{state['messages'][-1]}"
    )
    return {"messages": [response.content], "next_agent": "writer"}


def writer_agent(state: AgentState):
    """写手 Agent"""
    response = llm.invoke(
        f"作为技术写手，把研究内容写成文章：{state['messages'][-1]}"
    )
    return {"messages": [response.content], "next_agent": "reviewer"}


def reviewer_agent(state: AgentState):
    """审核 Agent"""
    response = llm.invoke(
        f"作为审核员，审阅这篇文章并给改进建议：{state['messages'][-1]}"
    )
    return {"messages": [response.content], "next_agent": END}


# 工作流
workflow = StateGraph(AgentState)
workflow.add_node("researcher", research_agent)
workflow.add_node("writer", writer_agent)
workflow.add_node("reviewer", reviewer_agent)

workflow.set_entry_point("researcher")
workflow.add_edge("researcher", "writer")
workflow.add_edge("writer", "reviewer")
workflow.add_edge("reviewer", END)

app = workflow.compile()

# 使用
result = app.invoke({
    "messages": ["写一篇关于 RAG 技术的科普文章"],
    "next_agent": "researcher",
})
print(result["messages"][-1])
```

```python
# 11_finetune_lora.py - 用 LoRA 微调模型
from peft import LoraConfig, get_peft_model, TaskType
from transformers import AutoModelForCausalLM, AutoTokenizer, TrainingArguments, Trainer
from datasets import load_dataset

# 1. 加载基模型
model = AutoModelForCausalLM.from_pretrained(
    "Qwen/Qwen2.5-7B-Instruct",
    load_in_4bit=True,  # 4bit 量化
    device_map="auto",
)
tokenizer = AutoTokenizer.from_pretrained("Qwen/Qwen2.5-7B-Instruct")

# 2. LoRA 配置
lora_config = LoraConfig(
    r=16,                # LoRA 秩
    lora_alpha=32,       # 缩放因子
    target_modules=["q_proj", "v_proj"],  # 目标层
    lora_dropout=0.05,
    bias="none",
    task_type=TaskType.CAUSAL_LM,
)

model = get_peft_model(model, lora_config)
model.print_trainable_parameters()
# trainable params: 4M || all params: 7B || trainable: 0.057%

# 3. 准备数据
dataset = load_dataset("your_dataset", split="train")

def tokenize(example):
    return tokenizer(
        example["text"],
        truncation=True,
        max_length=512,
    )

tokenized_dataset = dataset.map(tokenize)

# 4. 训练
training_args = TrainingArguments(
    output_dir="./lora_output",
    num_train_epochs=3,
    per_device_train_batch_size=4,
    learning_rate=2e-4,
    save_steps=100,
)

trainer = Trainer(
    model=model,
    args=training_args,
    train_dataset=tokenized_dataset,
)

trainer.train()

# 5. 保存
model.save_pretrained("./lora_final")
```

### 5.1 场景 1：搭建客户内部知识库 RAG

**做法**：
```python
# build_customer_kb.py
from langchain_community.document_loaders import (
    DirectoryLoader,
    UnstructuredWordDocumentLoader,
    UnstructuredPDFLoader,
)
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain_openai import OpenAIEmbeddings
from langchain_community.vectorstores import Milvus
from langchain.retrievers import ContextualCompressionRetriever
from langchain.retrievers.document_compressors import CohereRerank


def build_kb(docs_dir: str, milvus_host: str):
    """构建客户知识库"""

    # 1. 多格式文档加载
    loaders = [
        DirectoryLoader(docs_dir, glob="**/*.pdf", loader_cls=UnstructuredPDFLoader),
        DirectoryLoader(docs_dir, glob="**/*.docx", loader_cls=UnstructuredWordDocumentLoader),
        DirectoryLoader(docs_dir, glob="**/*.txt"),
    ]

    all_docs = []
    for loader in loaders:
        all_docs.extend(loader.load())

    print(f"加载了 {len(all_docs)} 份文档")

    # 2. 智能分块（按章节、段落）
    text_splitter = RecursiveCharacterTextSplitter(
        chunk_size=500,
        chunk_overlap=100,
        separators=["\n\n", "\n", "。", "！", "？", " ", ""],
    )
    chunks = text_splitter.split_documents(all_docs)
    print(f"分块后：{len(chunks)} 个 chunk")

    # 3. Embedding + 向量库（用 Milvus，支持大规模）
    embeddings = OpenAIEmbeddings(model="text-embedding-3-small")
    vectorstore = Milvus.from_documents(
        chunks,
        embeddings,
        connection_args={"host": milvus_host, "port": "19530"},
        collection_name="customer_kb",
    )

    # 4. 混合检索 + Rerank
    bm25_retriever = BM25Retriever.from_documents(chunks)
    vector_retriever = vectorstore.as_retriever(search_kwargs={"k": 10})
    ensemble_retriever = EnsembleRetriever(
        retrievers=[bm25_retriever, vector_retriever],
        weights=[0.3, 0.7],
    )

    rerank_compressor = CohereRerank(top_n=3)
    final_retriever = ContextualCompressionRetriever(
        base_compressor=rerank_compressor,
        base_retriever=ensemble_retriever,
    )

    return final_retriever


# 使用
retriever = build_kb("./客户文档/", "milvus.internal")
results = retriever.get_relevant_documents("年假政策")
```

**遇到的问题 & 解决**：
- 文档格式混乱 → 用 Unstructured 库
- 大文件加载慢 → 多进程并行
- 检索不准 → Hybrid + Rerank
- **总耗时：3 周**（含数据清洗）

### 6.1 坑 1：忽视 Prompt 调试

**症状**：
- 改一个 Prompt 词，效果差很多
- 不知道是 Prompt 问题还是模型问题

**避坑**：
- 用版本管理（LangSmith、PromptLayer）
- 用评估集（见 2.1.8）
- 每次改动跑评估
- 记录效果变化

### 6.2 坑 2：RAG 效果差就调 Prompt

**症状**：
- RAG 答非所问
- 一味改 Prompt，**实际问题在检索**

**避坑**：
- 先看检索结果（不是 LLM 输出）
- 检索对了但 LLM 答错 → 改 Prompt
- 检索错了 → 改 Embedding / 分块 / 检索策略
- 用 Rerank 提升精度

### 6.3 坑 3：Agent 无限循环

**症状**：
- Agent 一直调用工具不返回
- Token 烧光

**避坑**：

### 6.5 坑 5：Fine-tuning 当银弹

**症状**：

## 十一、高级 RAG 技术

### 11.1 ColBERT（细粒度检索）

> 传统向量检索：查询向量 vs 文档向量 → 单一相似度
> ColBERT：每个 token 都做相似度计算 → 细粒度匹配

**优势**：
- 召回率提升 5-15%
- 适合长文档、复杂问题
- 适合专业领域（法律、医疗）

**实现**：
```python
# colbert_search.py
from colbert import ColBERT
import numpy as np


class ColBERTRetriever:
    """ColBERT 检索器"""

    def __init__(self):
        self.colbert = ColBERT.from_pretrained(
            "colbert-ir/colbertv2.0",
        )
        self.documents = []  # 文档库

    def index_documents(self, docs: list):
        """索引文档"""
        self.documents = docs
        # 实际用 ColBERT 内部做 indexing
        # self.colbert.index(docs)

    def search(self, query: str, top_k: int = 3) -> list:
        """检索"""
        # 1. Query 编码（每个 token 一个向量）
        query_embeddings = self.colbert.encode_query(query)

        # 2. 与每个文档的 token 计算相似度（MaxSim）
        results = []
        for doc in self.documents:
            doc_embeddings = self.colbert.encode_doc(doc)
            # MaxSim: 每个 query token 与最相似的 doc token
            sim_matrix = np.dot(
                query_embeddings,
                doc_embeddings.T,
            )
            max_sim = sim_matrix.max(axis=1).sum()
            results.append((doc, max_sim))

        # 3. 排序返回
        results.sort(key=lambda x: -x[1])
        return results[:top_k]
```

### 11.2 Self-RAG（自我反思）

> 让 LLM 自我评估检索质量，必要时重新检索

```python
# self_rag.py - Self-RAG 实现
class SelfRAG:
    """Self-RAG：自我反思的 RAG"""

    def __init__(self, llm, retriever):
        self.llm = llm
        self.retriever = retriever
        self.reflect_prompt = """
判断以下检索结果是否足够回答问题。

问题：{question}
检索内容：{context}

回答格式：
- relevance: [yes/partial/no]
- is_supportive: [yes/no]
- confidence: [0-1]
"""

    async def query(self, question: str) -> str:
        """查询（带反思）"""
        # 1. 第一次检索
        contexts = await self.retrieve(question)

        # 2. 评估检索质量
        reflection = await self.reflect(question, contexts)

        # 3. 如果不相关，重新检索
        if "no" in reflection["relevance"]:
            # 用扩展后的 query 重新检索
            expanded = await self.expand_query(question)
            contexts = await self.retrieve(expanded)

        # 4. 生成答案
        answer = await self.generate(question, contexts)
        return answer

    async def reflect(self, question, contexts):
        """反思检索质量"""
        prompt = self.reflect_prompt.format(
            question=question,
            context="\n".join(contexts),
        )
        result = await self.llm.chat([{
            "role": "user", "content": prompt,
        }])
        return parse_reflection(result)
```

### 11.3 CRAG（Corrective RAG）

> 检索 → 评估 → 修正 → 重新生成

```python
# crag.py
class CRAG:
    """Corrective RAG"""

    async def query(self, question: str) -> str:
        # 1. 检索
        docs = await self.retrieve(question)

        # 2. 评估每个 doc
        relevant_docs = []
        for doc in docs:
            score = await self.grade_doc(question, doc)
            if score > 0.7:
                relevant_docs.append(doc)
            elif score > 0.3:
                # 模糊相关，用 LLM 改写
                rewritten = await self.rewrite_doc(doc, question)
                relevant_docs.append(rewritten)

        # 3. 如果都不相关，触发 Web 搜索
        if not relevant_docs:
            relevant_docs = await self.web_search(question)

        # 4. 生成
        return await self.generate(question, relevant_docs)

    async def grade_doc(self, question, doc) -> float:
        """给文档打分（0-1）"""
        prompt = f"""
判断文档与问题的相关度（0-1）。

问题：{question}
文档：{doc}

只输出分数："""
        result = await self.llm.chat([{
            "role": "user", "content": prompt,
        }])
        return float(result.strip())
```

### 11.4 HyDE（Hypothetical Document Embeddings）

> 用 LLM 生成假设性答案，用假设答案的 Embedding 去检索

**适用**：问题与文档用词差异大时

```python
# hyde.py
class HyDE:
    def __init__(self, llm, embedder, retriever):
        self.llm = llm
        self.embedder = embedder
        self.retriever = retriever

    async def query(self, question: str) -> str:
        # 1. 生成假设性答案
        hypothetical = await self.generate_hypothetical(question)

        # 2. 用假设答案检索
        query_vector = await self.embedder.embed(hypothetical)
        docs = await self.retriever.search(query_vector)

        # 3. 用实际 question + 检索内容生成
        return await self.generate(question, docs)

    async def generate_hypothetical(self, question: str) -> str:
        """生成假设性答案"""
        prompt = f"""
基于你的知识，写一段假设性的答案（100-200 字）。

问题：{question}
假设答案："""
        return await self.llm.chat([{
            "role": "user", "content": prompt,
        }])
```

```python
        self.cache = {}  # Context 缓存

        """上下文压缩（关键能力）"""

        # 1. 抽取关键信息
```

## 十三、Agent 设计模式进阶

### 13.1 ReWOO（Reasoning Without Observation）

> ReAct 的进化版：先规划，再执行，最后总结

```python
# rewoo.py
class ReWOOAgent:
    """ReWOO Agent"""

    async def solve(self, task: str) -> str:
        # 1. Planner：规划步骤（一次性生成所有步骤）
        plan = await self.plan(task)
        # plan = [
        #     {"step": 1, "tool": "search_docs", "input": "..."},
        #   {"step": 2, "tool": "calculator", "input": "..."},
        # ]

        # 2. Worker：执行所有步骤
        results = {}
        for step in plan:
            result = await self.execute_tool(
                step["tool"],
                step["input"],
            )
            results[step["step"]] = result

        # 3. Solver：基于所有结果总结
        return await self.solve(task, results)
```

**优势**：比 ReAct 更快（一次规划，多次执行），节省 token

### 13.2 Reflexion（自我反思）

```python
# reflexion.py
class ReflexionAgent:
    """带反思的 Agent"""

    async def solve(self, task: str, max_attempts: int = 3) -> str:
        reflections = []

        for attempt in range(max_attempts):
            # 1. 尝试解答
            answer = await self.attempt(task, reflections)

            # 2. 自我评估
            evaluation = await self.evaluate(answer, task)
            if evaluation["is_correct"]:
                return answer

            # 3. 反思失败原因
            reflection = await self.reflect(answer, evaluation)
            reflections.append(reflection)

        return answer  # 最后一次尝试
```

### 13.3 Plan-and-Execute

```python
# plan_execute.py
class PlanExecuteAgent:
    """规划 + 执行"""

    async def solve(self, task: str) -> str:
        # 1. 生成完整计划
        plan = await self.create_plan(task)
        # plan = ["搜索文档", "分析结果", "撰写报告"]

        # 2. 按计划执行
        results = []
        for step in plan:
            result = await self.execute_step(step, results)
            results.append(result)

            # 动态调整计划
            if self.should_replan(results):
                plan = await self.replan(task, results, plan)

        # 3. 综合结果
        return await self.synthesize(results)
```

```python
# lats.py
class LATSAgent:
    """树搜索 Agent"""

    async def solve(self, task: str) -> str:
        # 1. 构建搜索树
        root = Node(state=task)

        for _ in range(max_depth):
            # 2. 选择最有希望的节点
            node = self.select(root)

            # 3. 扩展（执行下一步）
            children = await self.expand(node)

            # 4. 评估
            for child in children:
                child.score = await self.evaluate(child)

            # 5. 反向传播
            self.backpropagate(node, node.score)

        # 6. 返回最佳路径
        return self.get_best_path(root)
```

```python
        }

    """客服 Agent"""
    def __init__(self):
        self.llm = ChatOpenAI(model="gpt-4o-mini")

    async def handle(self, state: AgentState) -> AgentState:
        # 情感分析
        sentiment = await analyze_sentiment(state["messages"][-1])
        response = await self.llm.ainvoke(
            f"用户情感：{sentiment}，作为客服回应：{state['messages'][-1]}"
        )
        return {
            "messages": [response.content],
            "next_agent": "manager",
        }


# Manager Agent（路由）
class ManagerAgent:
    def __init__(self):
        self.llm = ChatOpenAI(model="gpt-4o")
        self.agents = {
            "sales": SalesAgent(),
            "tech": TechAgent(),
            "support": SupportAgent(),
        }

    async def route(self, state: AgentState) -> str:
        """路由到合适的 Agent"""
        prompt = f"""判断用户问题应该路由到哪个 Agent：
- sales: 销售相关（报价、合同）
- tech: 技术问题（功能、bug）
- support: 客服（投诉、建议）

问题：{state['messages'][-1]}

只输出：sales / tech / support"""
        result = await self.llm.ainvoke(prompt)
        return result.content.strip()

    async def handle(self, state: AgentState) -> AgentState:
        # 1. 路由
        next_agent = await self.route(state)
        state["next_agent"] = next_agent
        return state


# 完整工作流
class MultiAgentSystem:
    def __init__(self):
        self.manager = ManagerAgent()
        self.workflow = self._build_workflow()

    def _build_workflow(self):
        workflow = StateGraph(AgentState)

        workflow.add_node("manager", self.manager.handle)
        workflow.add_node("sales", self.manager.agents["sales"].handle)
        workflow.add_node("tech", self.manager.agents["tech"].handle)
        workflow.add_node("support", self.manager.agents["support"].handle)

        workflow.set_entry_point("manager")

        # 条件路由
        def route_decision(state):
            return state["next_agent"]

        workflow.add_conditional_edges(
            "manager",
            route_decision,
            {
                "sales": "sales",
                "tech": "tech",
                "support": "support",
            },
        )

        # 回到 manager
        workflow.add_edge("sales", "manager")
        workflow.add_edge("tech", "manager")
        workflow.add_edge("support", "manager")

        # 结束条件
        def should_continue(state):
            if state.get("next_agent") == "end":
                return END
            return "manager"

        workflow.add_conditional_edges(
            "manager",
            should_continue,
            {"manager": "manager", END: END},
        )

        return workflow.compile()

    async def chat(self, message: str) -> str:
        result = await self.workflow.ainvoke({
            "messages": [message],
            "next_agent": "manager",
            "context": {},
            "user_intent": "",
        })
        return result["messages"][-1]


# 使用
system = MultiAgentSystem()
response = await system.chat("你们的产品多少钱？")
# → 路由到 Sales Agent
```

## 十九、面试题精选

### Q1：RAG 的核心问题是什么？
**A**：检索质量。包括 embedding、分块、检索策略、rerank 等。RAG 效果 = 检索 × 生成，检索错了 LLM 答不对。

### Q2：Agent 和 RAG 区别？
**A**：
- RAG：检索 + 生成，固定流程
- Agent：规划 + 工具调用 + 反思，动态决策
- 简单问答用 RAG，复杂任务用 Agent

### Q3：Function Calling 和 Tool Use 关系？
**A**：本质一样。Function Calling 是 OpenAI 的叫法，Tool Use 是 Anthropic 的叫法。都是让 LLM 调用工具。

### Q4：幻觉怎么彻底解决？
**A**：不能彻底解决，只能减少：RAG + 引用 + 拒答 + 一致性 + 审计。

### Q5：Fine-tuning vs RAG 怎么选？
**A**：
- RAG：知识更新、检索准确
- Fine-tuning：风格、格式、特定任务
- 优先 RAG，不行再 Fine-tuning

```python
from authlib.integrations.requests_client import OAuth2Session
from flask import Flask, redirect, url_for, session, request

app = Flask(__name__)
app.secret_key = "random_secret"

# OAuth 配置
client_id = "your_client_id"
client_secret = "your_client_secret"
authorization_base_url = "https://客户SSO/oauth2/authorize"
token_url = "https://客户SSO/oauth2/token"
scope = ["openid", "profile", "email"]


@app.route("/login")
def login():
    """发起 OAuth 登录"""
    oauth = OAuth2Session(client_id, scope=scope)
    authorization_url, state = oauth.authorization_url(authorization_base_url)
    session["oauth_state"] = state
    return redirect(authorization_url)


@app.route("/callback")
def callback():
    """OAuth 回调"""
    oauth = OAuth2Session(client_id, state=session["oauth_state"])
    token = oauth.fetch_token(
        token_url,
        client_secret=client_secret,
        authorization_response=request.url,
    )
    session["oauth_token"] = token

    # 获取用户信息
    userinfo = oauth.get("https://客户SSO/oauth2/userinfo").json()
    return {"user": userinfo}
```

### 示例 3：OIDC（OpenID Connect）

```python
# 03_oidc.py - OIDC 集成
import httpx
from jose import jwt, jwk
from jose.utils import base64url_decode

# OIDC 配置
OIDC_ISSUER = "https://客户SSO"
OIDC_JWKS_URL = f"{OIDC_ISSUER}/.well-known/jwks.json"
OIDC_AUDIENCE = "ai-service"


def get_jwks():
    """获取 JWKS（公钥）"""
    response = httpx.get(OIDC_JWKS_URL)
    return response.json()


def verify_id_token(id_token: str) -> dict:
    """验证 ID Token"""
    # 1. 获取 JWKS
    jwks = get_jwks()

    # 2. 解码 header 获取 kid
    header = jwt.get_unverified_header(id_token)
    kid = header["kid"]

    # 3. 找公钥
    key = None
    for jwk_key in jwks["keys"]:
        if jwk_key["kid"] == kid:
            key = jwk.construct(jwk_key)
            break

    if not key:
        raise ValueError("未找到匹配的公钥")

    # 4. 验证签名
    payload = jwt.decode(
        id_token,
        key,
        algorithms=["RS256"],
        audience=OIDC_AUDIENCE,
        issuer=OIDC_ISSUER,
    )

    return payload
```

```python
        self.corp_id = corp_id

        params = {"corpid": self.corp_id, "corpsecret": self.corp_secret}
        response = httpx.get(url, params=params)
        data = response.json()
        if data["errcode"] == 0:
            self.access_token = data["access_token"]

        data = {
            "touser": "|".join(user_ids),
            "msgtype": "text",
            "agentid": self.agent_id,
            "text": {"content": content},
        }
        response = httpx.post(url, params=params, json=data)
```

### 示例 10：API 网关 / 限流

```python
# 10_api_gateway.py - API 网关 + 限流
import time
from fastapi import FastAPI, Request, HTTPException
from fastapi.responses import JSONResponse
import redis

app = FastAPI()
redis_client = redis.Redis()


# 限流装饰器
def rate_limit(max_requests: int = 100, window: int = 60):
    """滑动窗口限流"""
    def decorator(func):
        async def wrapper(request: Request, *args, **kwargs):
            # 1. 限流 key
            client_ip = request.client.host
            key = f"rate_limit:{client_ip}:{int(time.time() // window)}"

            # 2. 计数
            current = redis_client.incr(key)
            if current == 1:
                redis_client.expire(key, window)

            # 3. 超过限制
            if current > max_requests:
                raise HTTPException(429, "请求过于频繁")

            return await func(request, *args, **kwargs)
        return wrapper
    return decorator


# 全局限流中间件
@app.middleware("http")
async def rate_limit_middleware(request: Request, call_next):
    # 1. 跳过健康检查
    if request.url.path in ["/health", "/ready"]:
        return await call_next(request)

    # 2. 限流
    client_ip = request.client.host
    key = f"rate_limit:{client_ip}:{int(time.time() // 60)}"
    current = redis_client.incr(key)

    if current == 1:
        redis_client.expire(key, 60)

    if current > 1000:  # 每分钟 1000 次
        return JSONResponse(
            status_code=429,
            content={"detail": "Too many requests"},
        )

    # 3. 添加响应头
    response = await call_next(request)
    response.headers["X-RateLimit-Limit"] = "1000"
    response.headers["X-RateLimit-Remaining"] = str(1000 - current)
    return response


# API 路由
@app.post("/api/v1/query")
@rate_limit(max_requests=100, window=60)
async def query(req: QueryRequest):
    return await process_query(req)
```

### 5.1 场景 1：客户要求用 SSO 登录

**业务**：客户用 Azure AD，员工要用公司账号登录 AI 系统

**做法**：OIDC 集成（见示例 3）

**遇到的问题**：
- 客户的 OIDC 配置不规范（缺 JWKS） → 推动客户修复
- Token 过期时间短（5 分钟） → 用 refresh_token
- 多租户 → 用 tenant_id 区分
- **总耗时：2 周**

### 5.3 场景 3：客户企业微信集成

**业务**：客户用企微，想在企微里直接用 AI

**做法**：

```python
import xml.etree.ElementTree as ET

app = Flask(__name__)


@app.route("/webhook/wechat", methods=["POST"])
def wechat_webhook():
    """企微回调"""
```

## 十三、Webhook 安全

### 13.1 Webhook 签名验证

```python
# webhook_security.py
import hmac
import hashlib
import time
from fastapi import FastAPI, Request, HTTPException

app = FastAPI()

WEBHOOK_SECRET = "your-secret-key"


@app.post("/webhook")
async def webhook(request: Request):
    """接收 Webhook"""
    body = await request.body()
    signature = request.headers.get("X-Signature")
    timestamp = request.headers.get("X-Timestamp")

    # 1. 验证时间戳（防重放）
    if abs(time.time() - int(timestamp)) > 300:  # 5 分钟内
        raise HTTPException(status_code=401, detail="Timestamp too old")

    # 2. 验证签名
    expected = hmac.new(
        WEBHOOK_SECRET.encode(),
        f"{timestamp}.{body.decode()}".encode(),
        hashlib.sha256,
    ).hexdigest()

    if not hmac.compare_digest(signature, expected):
        raise HTTPException(status_code=401, detail="Invalid signature")

    # 3. 处理业务逻辑
    data = await request.json()
    await process_webhook(data)

    return {"status": "ok"}
```

### 13.2 发送方（签名）

```python
# webhook_sender.py
import hmac
import hashlib
import time
import httpx


async def send_webhook(url, payload, secret):
    """发送带签名的 Webhook"""
    timestamp = str(int(time.time()))
    body = json.dumps(payload)

    signature = hmac.new(
        secret.encode(),
        f"{timestamp}.{body}".encode(),
        hashlib.sha256,
    ).hexdigest()

    headers = {
        "X-Signature": signature,
        "X-Timestamp": timestamp,
        "Content-Type": "application/json",
    }

    async with httpx.AsyncClient() as client:
        response = await client.post(url, content=body, headers=headers)
        return response
```

```python
# event_sourcing.py
class EventStore:
    """事件存储"""

    def __init__(self):
        self.events = []  # 简化：实际用数据库

    def append(self, event):
        """追加事件"""
        event["version"] = len(self.events) + 1
        event["timestamp"] = time.time()
        self.events.append(event)

    def get_events(self, aggregate_id):
        """获取聚合的所有事件"""
        return [
            e for e in self.events
            if e.get("aggregate_id") == aggregate_id
        ]

    def rebuild_state(self, aggregate_id, initial_state):
        """从事件重建状态"""
        events = self.get_events(aggregate_id)
        state = initial_state
        for event in events:
            state = apply_event(state, event)
        return state
```

### 14.4 Saga 模式（分布式事务）

```python
# saga_pattern.py
class Saga:
    """Saga 分布式事务"""

    def __init__(self):
        self.steps = []
        self.compensations = []

    def add_step(self, action, compensation):
        """添加步骤和补偿"""
        self.steps.append(action)
        self.compensations.append(compensation)

    async def execute(self, context):
        """执行 Saga"""
        executed_steps = []

        try:
            for i, step in enumerate(self.steps):
                await step(context)
                executed_steps.append(i)

            return context

        except Exception as e:
            # 补偿：反向执行
            for i in reversed(executed_steps):
                await self.compensations[i](context)

            raise


# 实战：下单 + 支付 + 减库存
saga = Saga()
saga.add_step(
    action=create_order,
    compensation=delete_order,
)
saga.add_step(
    action=charge_payment,
    compensation=refund_payment,
)
saga.add_step(
    action=decrement_inventory,
    compensation=increment_inventory,
)

try:
    result = await saga.execute({"user_id": "u1", "items": [...]})
except:
    # 自动补偿（订单取消、退款、库存恢复）
    pass
```

```python
import time
import hmac
import hashlib
import base64


class DingTalkAdapter:
    """钉钉集成"""

    def __init__(self, app_key, app_secret):
        self.app_key = app_key
        self.app_secret = app_secret

        self.token_expires = 0

    async def get_access_token(self):
        """获取 access_token"""
        if time.time() < self.token_expires:
            return self.access_token

        url = "https://oapi.dingtalk.com/gettoken"
        params = {
            "appkey": self.app_key,
            "appsecret": self.app_secret,

        async with httpx.AsyncClient() as client:
            resp = await client.get(url, params=params)
            data = resp.json()
            self.access_token = data["access_token"]
            self.token_expires = time.time() + 7000

        return self.access_token

    async def send_message(self, user_ids: list, message: str):
        """发送消息"""
        token = await self.get_access_token()

            "agent_id": self.agent_id,

                "msgtype": "text",
                "text": {"content": message},
            },
        }

            resp = await client.post(
                f"{url}?access_token={token}",
```

```python
            "send_notification": self.send_notification,
        }

    async def handle_request(self, user_id: str, message: str) -> str:
        """处理用户请求"""
        # 1. RAG 检索（知识库）
        context = await self.rag.search(message)

        # 2. Agent 决策（是否需要调用工具）
        response = await self.agent.run(
            message=message,
            context=context,
            tools=self.tools,
        )

        return response

    async def query_order(self, order_id: str) -> dict:
        """查询订单（从 SAP）"""
        return await self.sap.get_sales_order(order_id)

    async def create_ticket(self, title: str, description: str) -> str:
        """创建工单（到 CRM）"""
        return await self.crm.create_ticket({
            "Subject": title,
            "Description": description,
        })

    async def send_notification(self, user_id: str, message: str):
        """发送通知（到钉钉）"""
        await self.dingtalk.send_message([user_id], message)
```

## 十八、故障排查手册

### 18.1 故障 1：API 调用超时

**症状**：客户调用 AI 服务超时

**排查**：
```bash
# 1. 网络
ping customer.com
traceroute api.customer.com

# 2. DNS
nslookup api.customer.com

# 3. 防火墙
nc -zv api.customer.com 443
```

**解决**：
- 调超时
- 加重试
- 异步化

### 18.2 故障 2：OAuth Token 失效

**症状**：401 Unauthorized

**解决**：
- 实现 Token 自动刷新
- 缓存 Token
- 重试逻辑

### 18.3 故障 3：WebHook 收不到

**排查**：
- 检查防火墙
- 验证签名
- 看重试日志

### 18.4 故障 4：SAP 连接失败

**症状**：pyrfc 连接错误

**解决**：
- 检查 SAP 网关

- 看 SAP 路由

### 18.5 故障 5：Kafka 消费堆积

**症状**：Consumer 跟不上

**解决**：
- 加 consumer 实例
- 优化消费逻辑
- 调 batch size

### 18.6 故障 6：数据不一致

**症状**：两边数据不同步

**解决**：
- 加校验任务

### 18.7 故障 7：限流被误伤

**症状**：用户请求被拒

**解决**：

- 多级限流

## 二十、面试题精选

### Q1：REST vs GraphQL vs gRPC 选型？
**A**：
- REST：通用、简单、缓存友好
- GraphQL：复杂查询、按需取字段
- gRPC：高性能、强类型、双向流

### Q2：OAuth 2.0 流程？
**A**：
- 授权码模式（最常用）
- 客户端凭证模式（服务间）
- 简化模式（公开 API）
- 密码模式（已不推荐）

### Q3：什么是 Webhook？vs 轮询？
**A**：
- Webhook：服务端推送
- 轮询：客户端定时拉
- Webhook 实时、省资源

### Q4：什么是事件驱动？
**A**：
- 服务间通过事件解耦
- 异步、最终一致
- 用 Kafka / RabbitMQ

### Q5：Idempotency Key 是什么？
**A**：
- 唯一标识请求
- 防止重复处理
- 关键：业务侧实现

### Q6：什么是 Saga？
**A**：
- 分布式事务
- 一系列本地事务 + 补偿
- 最终一致

### Q7：JWT 和 Session 区别？
**A**：
- JWT：无状态、自包含
- Session：有状态、需存储
- JWT 适合分布式

### Q8：API 版本管理策略？
**A**：
- URL 版本（/v1/）
- Header 版本
- 参数版本
- 媒体类型

```python
# decorator.py - 用装饰器给函数计时
import time
from functools import wraps


def timer(func):
    """计时装饰器：FDE 必备的调试工具"""
    @wraps(func)
    def wrapper(*args, **kwargs):
        start = time.perf_counter()
        result = func(*args, **kwargs)
        elapsed = time.perf_counter() - start
        print(f"[{func.__name__}] 耗时 {elapsed:.2f}s")
        return result
    return wrapper


@timer
def call_llm(question: str) -> str:
    """调用 LLM"""
    from openai import OpenAI
    client = OpenAI()
    response = client.chat.completions.create(
        model="gpt-4o-mini",
        messages=[{"role": "user", "content": question}],
    )
    return response.choices[0].message.content


# 使用
answer = call_llm("什么是 RAG？")
# [call_llm] 耗时 1.23s
```

```python
import asyncio
from openai import AsyncOpenAI

client = AsyncOpenAI()


async def call_llm_async(question: str) -> str:
    """异步调用 LLM"""
    response = await client.chat.completions.create(
        model="gpt-4o-mini",
        messages=[{"role": "user", "content": question}],

    return response.choices[0].message.content

    """并发批量调用"""
    tasks = [call_llm_async(q) for q in questions]
    return await asyncio.gather(*tasks)
```

```python
from contextlib import contextmanager
import psycopg


@contextmanager
def db_connection(dsn: str):
    """数据库连接上下文管理器"""
    conn = psycopg.connect(dsn)
    try:
        yield conn
    finally:
        conn.close()
```

```python
def read_large_file(path: str):
    """逐行读取大文件（不占内存）"""
    with open(path, "r", encoding="utf-8") as f:
        for line in f:
            yield line.strip()


def process_data(lines):
    """流式处理"""
```

```python
from pydantic import BaseModel, Field
from openai import OpenAI
import json

client = OpenAI()


class UserInfo(BaseModel):
    """用户信息结构"""

    city: str = Field(..., description="城市")
```

### 4.8 示例 8：FastAPI RAG 服务（完整生产级）

```python
# rag_service.py - FDE 必会的 FastAPI RAG 服务
from fastapi import FastAPI, HTTPException
from pydantic import BaseModel
from openai import OpenAI
import chromadb

app = FastAPI(title="RAG Service")
client = OpenAI()

# 向量库
chroma = chromadb.PersistentClient(path="./chroma_db")
collection = chroma.get_or_create_collection("docs")


class QueryRequest(BaseModel):
    question: str
    top_k: int = 3


class QueryResponse(BaseModel):
    answer: str
    sources: list


@app.post("/query", response_model=QueryResponse)
def query(req: QueryRequest):
    """RAG 查询接口"""
    # 1. 检索
    results = collection.query(
        query_texts=[req.question],
        n_results=req.top_k,
    )

    if not results["documents"]:
        raise HTTPException(404, "无相关内容")

    # 2. 构造 Prompt
    context = "\n\n".join(results["documents"][0])
    prompt = f"""基于以下参考资料回答问题。如果资料中没有答案，请说"我不知道"。

参考资料：
{context}

问题：{req.question}

回答："""

    # 3. 调用 LLM
    response = client.chat.completions.create(
        model="gpt-4o-mini",
        messages=[{"role": "user", "content": prompt}],
    )

    return QueryResponse(
        answer=response.choices[0].message.content,
        sources=results["documents"][0],
    )


# 启动：uvicorn rag_service:app --reload
```

### 4.9 示例 9：SQL 复杂查询（业务必备）

```sql
-- customer_analysis.sql - 客户分析（窗口函数 + CTE）
WITH monthly_revenue AS (
    SELECT
        customer_id,
        DATE_TRUNC('month', order_date) AS month,
        SUM(amount) AS revenue
    FROM orders
    WHERE order_date >= '2025-01-01'
    GROUP BY customer_id, DATE_TRUNC('month', order_date)
),
customer_stats AS (
    SELECT
        customer_id,
        SUM(revenue) AS total_revenue,
        COUNT(DISTINCT month) AS active_months,
        AVG(revenue) AS avg_monthly_revenue
    FROM monthly_revenue
    GROUP BY customer_id
)
SELECT
    c.id,
    c.name,
    cs.total_revenue,
    cs.active_months,
    cs.avg_monthly_revenue,
    RANK() OVER (ORDER BY cs.total_revenue DESC) AS revenue_rank,
    NTILE(4) OVER (ORDER BY cs.total_revenue DESC) AS revenue_quartile
FROM customers c
JOIN customer_stats cs ON c.id = cs.customer_id
WHERE cs.active_months >= 3  -- 至少活跃 3 个月
ORDER BY cs.total_revenue DESC
LIMIT 100;
```

### 4.11 示例 11：日志 + 异常处理（生产级）

```python
# production_code.py - 生产级代码模板
import logging
from functools import wraps


# 配置日志
logging.basicConfig(
    level=logging.INFO,
    format="%(asctime)s [%(levelname)s] %(name)s: %(message)s",
)
logger = logging.getLogger(__name__)


def retry(max_attempts: int = 3, delay: float = 1.0):
    """重试装饰器：LLM 调用必备"""
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            for attempt in range(1, max_attempts + 1):
                try:
                    return func(*args, **kwargs)
                except Exception as e:
                    if attempt == max_attempts:
                        logger.error(f"[{func.__name__}] 最终失败：{e}")
                        raise
                    logger.warning(
                        f"[{func.__name__}] 第{attempt}次失败：{e}, "
                        f"{delay}s 后重试"
                    )
                    import time
                    time.sleep(delay * attempt)  # 指数退避
        return wrapper
    return decorator


@retry(max_attempts=3, delay=2.0)
def call_llm_with_retry(prompt: str) -> str:
    """带重试的 LLM 调用"""
    from openai import OpenAI
    client = OpenAI()
    response = client.chat.completions.create(
        model="gpt-4o-mini",
        messages=[{"role": "user", "content": prompt}],
    )
    return response.choices[0].message.content


# 使用
try:
    answer = call_llm_with_retry("什么是 RAG？")
    logger.info(f"成功：{answer[:50]}")
except Exception as e:
    logger.error(f"调用失败：{e}")
```

```python
async def get_current_user(token: str = Depends(oauth2_scheme)):

@app.get("/api/me")
```

```python
import asyncio
import aiohttp
import aiokafka


async def consume_logs():
    """消费 Kafka 日志"""
```

```python
# descriptor.py - 描述符基础
class TypedAttribute:
    """类型校验描述符"""
    def __init__(self, name, type_, min_value=None, max_value=None):
        self.name = name
        self.type_ = type_
        self.min_value = min_value
        self.max_value = max_value

    def __get__(self, instance, owner):
        if instance is None:
            return self
        return instance.__dict__.get(self.name)

    def __set__(self, instance, value):
        if not isinstance(value, self.type_):
            raise TypeError(f"{self.name} must be {self.type_}")
        if self.min_value is not None and value < self.min_value:
            raise ValueError(f"{self.name} must be >= {self.min_value}")
        if self.max_value is not None and value > self.max_value:
            raise ValueError(f"{self.name} must be <= {self.max_value}")
        instance.__dict__[self.name] = value


class Customer:
    name = TypedAttribute("name", str)
    age = TypedAttribute("age", int, min_value=0, max_value=150)
    score = TypedAttribute("score", float, min_value=0.0, max_value=100.0)

    def __init__(self, name: str, age: int, score: float):
        self.name = name
        self.age = age
        self.score = score


# 使用
```

```python
# metaclass.py - 用元类实现 ORM
class ModelMeta(type):
    """ORM 元类：自动收集字段"""
    def __new__(mcs, name, bases, attrs):
        # 收集所有字段
        fields = {}
        for key, value in list(attrs.items()):
            if isinstance(value, Field):
                fields[key] = value
                value.name = key

        # 替换为 descriptor
        for key, value in fields.items():
            attrs[key] = value

        attrs["_fields"] = fields
        return super().__new__(mcs, name, bases, attrs)


class Field:
    """字段基类"""
    def __init__(self, type_, required=True):
        self.type_ = type_
        self.required = required
        self.name = None


class StringField(Field):
    def __init__(self, max_length=255, **kwargs):
        super().__init__(str, **kwargs)
        self.max_length = max_length


class IntegerField(Field):
    def __init__(self, **kwargs):
        super().__init__(int, **kwargs)


# 使用元类
class BaseModel(metaclass=ModelMeta):
    def to_dict(self):
        return {k: getattr(self, k) for k in self._fields}


class User(BaseModel):
    name = StringField(max_length=50)
    age = IntegerField()


u = User()
u.name = "张三"
u.age = 30
print(u.to_dict())
# {'name': '张三', 'age': 30}
```

### 13.3 LLM 封装

```python
# app/services/llm.py
import logging
from openai import AsyncOpenAI, OpenAIError, RateLimitError
from tenacity import (
    retry, stop_after_attempt,
    wait_exponential, retry_if_exception_type,
)

from app.config import settings

logger = logging.getLogger(__name__)


class LLMService:
    """LLM 服务封装"""

    def __init__(self):
        self.client = AsyncOpenAI(api_key=settings.OPENAI_API_KEY)
        self.model = settings.LLM_MODEL

    @retry(
        stop=stop_after_attempt(3),
        wait=wait_exponential(min=1, max=10),
        retry=retry_if_exception_type((RateLimitError, OpenAIError)),
    )
    async def chat(
        self,
        messages: list[dict],
        temperature: float = None,
        max_tokens: int = None,
    ) -> str:
        """聊天接口（带重试）"""
        try:
            response = await self.client.chat.completions.create(
                model=self.model,
                messages=messages,
                temperature=temperature or settings.LLM_TEMPERATURE,
                max_tokens=max_tokens or settings.LLM_MAX_TOKENS,
            )
            return response.choices[0].message.content

        except RateLimitError as e:
            logger.warning(f"限流：{e}")
            raise  # 重试

        except OpenAIError as e:
            logger.error(f"OpenAI 错误：{e}")
            raise

    async def stream_chat(self, messages: list[dict]):
        """流式聊天"""
        stream = await self.client.chat.completions.create(
            model=self.model,
            messages=messages,
            stream=True,
        )
        async for chunk in stream:
            if chunk.choices[0].delta.content:
                yield chunk.choices[0].delta.content
```

### 13.4 RAG 核心

```python
# app/services/rag.py
import logging
import time
from typing import List

from app.services.llm import LLMService
from app.services.embedding import EmbeddingService
from app.services.vector_store import VectorStoreService
from app.models.schemas import QueryRequest, QueryResponse, Source

logger = logging.getLogger(__name__)


class RAGService:
    """RAG 服务"""

    def __init__(self):
        self.llm = LLMService()
        self.embedder = EmbeddingService()
        self.vector_store = VectorStoreService()
        self.prompt_template = self._build_prompt()

    def _build_prompt(self) -> str:
        """构建 Prompt 模板"""
        return """你是一个专业的助手。基于以下参考资料回答用户问题。

## 严格要求
1. 严格基于参考资料回答，不要编造
2. 如果资料中没有答案，请明确说"根据资料无法回答"
3. 回答要简洁、准确、有引用（用 [1][2] 标注）

## 参考资料
{context}

## 用户问题
{question}

## 回答
"""

    async def query(
        self,
        request: QueryRequest,
    ) -> QueryResponse:
        """RAG 查询"""
        start = time.time()

        # 1. 检索
        sources = await self.retrieve(
            request.question,
            top_k=request.top_k,
        )

        # 2. 构造 Prompt
        context = self._format_context(sources)
        prompt = self.prompt_template.format(
            context=context,
            question=request.question,
        )

        # 3. 生成
        answer = await self.llm.chat([{
            "role": "user",
            "content": prompt,
        }])

        latency = (time.time() - start) * 1000

        return QueryResponse(
            question=request.question,
            answer=answer,
            sources=[
                Source(content=s["content"], score=s["score"])
                for s in sources
            ],
            latency_ms=latency,
        )

    async def retrieve(
        self,
        question: str,
        top_k: int = 3,
    ) -> List[dict]:
        """检索相关文档"""
        # 1. Embedding
        query_vector = await self.embedder.embed(question)

        # 2. 向量检索
        results = await self.vector_store.search(
            query_vector=query_vector,
            top_k=top_k,
        )
        return results

    def _format_context(self, sources: List[dict]) -> str:
        """格式化上下文"""
        parts = []
        for i, s in enumerate(sources, 1):
            parts.append(f"[{i}] {s['content']}")
        return "\n\n".join(parts)
```

## 十六、故障排查手册（FDE 救命手册）

### 16.1 故障 1：内存泄漏

**症状**：
- 服务运行越久，内存占用越高
- 最终 OOM 被杀

**排查**：
```python
# 1. 用 memory_profiler 定位
from memory_profiler import profile

@profile
def my_function():
    # 业务代码
    pass

# 2. 用 tracemalloc 看分配
import tracemalloc
tracemalloc.start()

# 业务代码

snapshot = tracemalloc.take_snapshot()
for stat in snapshot.statistics("lineno")[:10]:
    print(stat)
```

**解决**：
- 找到泄漏点（一般是缓存无界增长）
- 加 LRU 限制
- 用 `__slots__`
- 用弱引用

### 16.2 故障 2：异步死锁

**症状**：
- 异步任务卡住，不返回
- 资源耗尽

**排查**：
```python
# 检查是否有未 await 的协程
# 错误：
async def bad():
    some_async_function()  # 没 await！

# 正确：
async def good():
    await some_async_function()
```

**解决**：
- 所有 async 函数都加 await
- 用 `asyncio.gather()` 收集结果
- 避免循环引用

### 16.3 故障 3：GIL 限制

**症状**：
- 多线程 Python 性能上不去
- CPU 密集任务慢

**解决**：
- 用 multiprocessing 替代 threading
- 用 C 扩展（numpy、Cython）
- 关键路径用 C++/Rust

### 16.4 故障 4：SQL 慢查询

**症状**：
- 接口超时
- 数据库 CPU 100%

**排查**：
```sql
-- 1. 找慢查询
SELECT * FROM pg_stat_statements
ORDER BY total_time DESC LIMIT 10;

-- 2. 分析执行计划
EXPLAIN ANALYZE SELECT ...;

-- 3. 加索引
CREATE INDEX ...;
```

### 16.5 故障 5：连接池耗尽

**症状**：
- "connection timeout"
- 服务挂起

**解决**：
- 增加连接池大小
- 检查连接泄漏（用 finally 关闭）
- 加超时

## 十九、面试题精选（编程语言）

### Q1：Python 的 GIL 是什么？多线程真的没用吗？
**A**：GIL 是全局解释器锁，保证同一时刻只有一个线程执行 Python 字节码。**CPU 密集任务**多线程确实没用，**IO 密集任务**多线程仍有效（因为 IO 等待时释放 GIL）。解决：用 multiprocessing（多进程）或 asyncio。

### Q2：async/await 和 threading 区别？
**A**：
- **threading**：多线程，OS 调度，适合 IO 密集
- **async/await**：单线程协程，用户态调度，适合高并发 IO
- 性能：async/await 通常比 threading 高（无线程切换开销）

### Q3：Python 内存管理？
**A**：引用计数 + 循环检测 GC。引用计数为主，循环引用靠 GC 回收。可以用 `gc.collect()` 手动触发。

### Q4：__new__ 和 __init__ 区别？
**A**：
- `__new__`：创建实例（类方法），返回实例
- `__init__`：初始化实例（实例方法）
- 典型应用：`__new__` 用于单例、元类

### Q5：SQL 慢查询怎么排查？
**A**：
1. 查慢查询日志（`pg_stat_statements`）
2. `EXPLAIN ANALYZE` 看执行计划
3. 加索引 / 优化 SQL
4. 考虑分库分表

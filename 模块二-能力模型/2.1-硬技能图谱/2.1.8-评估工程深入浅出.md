## 一、一句话定义

**评估工程 = 用数据证明 AI 有效的能力**。FDE 现场常被问"AI 真的有用吗？"，**没有评估 = 答不上来 = 失去信任**。

## 二、为什么 FDE 必须学

### 2.1 评估是 FDE 的"基本功"

> 没有评估 = 没有改进
> 没有评估 = 无法证明价值
> 没有评估 = 无法对比方案
> 没有评估 = 无法说服客户

### 2.2 评估的 4 个层级

```
┌────────────────────────────────────────┐
│ Level 1: 模型层（算法主导）              │
│  ├─ 准确率、F1、Recall                │
│  └─ 幻觉率、拒答率                    │
├────────────────────────────────────────┤
│ Level 2: 应用层（FDE 主导）⭐           │
│  ├─ 任务完成率                        │
│  ├─ 用户满意度                        │
│  └─ 转人工率                          │
├────────────────────────────────────────┤
│ Level 3: 业务层（业务方主导）           │
│  ├─ 转化率变化                        │
│  ├─ 成本降低                          │
│  └─ 效率提升                          │
├────────────────────────────────────────┤
│ Level 4: 用户体验（持续监测）           │
│  ├─ NPS                              │
│  ├─ 满意度                            │
│  └─ 留存率                            │
└────────────────────────────────────────┘
```

## 三、核心概念图解

### 3.1 评估工作流

```
[1] 准备评估集
     ↓
[2] 运行目标系统
     ↓
[3] 计算指标
     ├─ 自动化（DeepEval/RAGAS）
     ├─ LLM-as-Judge
     └─ 人工抽检
     ↓
[4] 输出报告
     ↓
[5] 改进 → 回到 [1]
```

### 3.2 评估指标体系

```
技术指标
  ├─ 准确率 / 召回率
  ├─ 幻觉率
  ├─ 延迟
  └─ 成本

业务指标
  ├─ 转化率
  ├─ 用户满意度
  ├─ 转人工率
  └─ ROI

A/B 测试
  ├─ 新旧对比
  ├─ 多版本对比
  └─ 显著性检验
```

### 3.3 评估 vs 监控

```
评估（Evaluation）
  ├─ 离线的、批量的
  ├─ 用评估集
  ├─ 适合：模型升级、Prompt 优化
  └─ 输出：报告

监控（Monitoring）
  ├─ 实时的、持续的
  ├─ 用生产数据
  ├─ 适合：异常告警
  └─ 输出：告警
```

```python
# 04_ragas.py - RAGAS 评估 RAG
from ragas import evaluate
from ragas.metrics import (
    faithfulness,           # 答案与检索内容一致性
    answer_relevancy,      # 答案与问题相关度
    context_precision,     # 检索上下文精确度
    context_recall,        # 检索上下文召回率
)
from datasets import Dataset


def evaluate_rag(test_cases: list) -> dict:
    """评估 RAG 系统"""
    # 1. 准备数据
    questions = []
    answers = []
    contexts = []
    ground_truths = []

    for case in test_cases:
        result = rag_query(case["question"])

        questions.append(case["question"])
        answers.append(result["answer"])
        contexts.append(result["retrieved_docs"])
        ground_truths.append(case["expected_answer"])

    # 2. 构造 Dataset
    dataset = Dataset.from_dict({
        "question": questions,
        "answer": answers,
        "contexts": contexts,
        "ground_truth": ground_truths,
    })

    # 3. 评估
    result = evaluate(
        dataset,
        metrics=[
            faithfulness,
            answer_relevancy,
            context_precision,
            context_recall,
        ],
    )

    return dict(result)


# 使用
test_cases = [
    {
        "question": "公司年假？",
        "expected_answer": "员工每年10天年假",
    },
    {
        "question": "报销流程？",
        "expected_answer": "提交发票后3个工作日",
    },
    # ... 200 条
```

### 示例 5：LLM-as-Judge

```python
# 05_llm_judge.py - LLM 当评委
from openai import OpenAI

client = OpenAI()


def llm_judge(question: str, expected: str, actual: str, criteria: str) -> dict:
    """用 LLM 评估答案"""
    prompt = f"""你是一个严格的评估专家，根据标准给答案打分。

## 评估标准
{criteria}

## 问题
{question}

## 标准答案
{expected}

## 实际答案
{actual}

请按以下格式输出：
- 分数：0.X（0-1 之间）
- 理由：xxx
- 改进建议：xxx"""

    response = client.chat.completions.create(
        model="gpt-4o",
        messages=[{"role": "user", "content": prompt}],
        temperature=0,
    )

    # 解析分数
    content = response.choices[0].message.content
    score = 0.0
    for line in content.split("\n"):
        if "分数" in line:
            try:
                score = float(line.split("：")[-1].strip())
            except:
                pass

    return {
        "score": score,
        "reasoning": content,
    }


# 使用
result = llm_judge(
    question="公司年假？",
    expected="员工每年10天年假",
    actual="公司每年提供10天年假",
    criteria="答案是否准确、完整、自然",
)
print(result)
```

### 示例 6：多维度 LLM Judge

```python
# 06_multi_dim_judge.py - 多维度评估
class MultiDimJudge:
    """多维度 LLM Judge"""

    def __init__(self):
        self.client = OpenAI()

        self.dimensions = {
            "准确性": "答案与标准答案的事实一致性（0-1）",
            "完整性": "答案是否覆盖标准答案的所有要点（0-1）",
            "清晰度": "答案表达是否清晰、无歧义（0-1）",
            "专业性": "答案是否使用专业、准确的术语（0-1）",
            "格式": "答案格式是否符合要求（0-1）",
        }

    def evaluate(self, question: str, expected: str, actual: str) -> dict:
        """多维度评估"""
        results = {}
        for dim, criteria in self.dimensions.items():
            prompt = f"""评估「{dim}」：{criteria}

问题：{question}
标准答案：{expected}
实际答案：{actual}

只输出分数（0-1）："""
            response = self.client.chat.completions.create(
                model="gpt-4o-mini",
                messages=[{"role": "user", "content": prompt}],
                temperature=0,
            )
            try:
                score = float(response.choices[0].message.content.strip())
            except:
                score = 0.0
            results[dim] = score

        results["综合分"] = sum(results.values()) / len(results)
        return results


# 使用
judge = MultiDimJudge()
result = judge.evaluate(
    question="公司年假？",
    expected="员工每年10天年假",
    actual="公司每年提供10天年假",
)
print(result)
# {'准确性': 0.95, '完整性': 0.90, '清晰度': 0.85,
#  '专业性': 0.80, '格式': 0.75, '综合分': 0.85}
```

```python
# 08_ab_test.py - A/B 测试
import hashlib
import random


class ABTestRouter:
    """A/B 测试路由器"""

    def __init__(self, experiments: dict):
        self.experiments = experiments

    def assign_variant(self, user_id: str, experiment_name: str) -> str:
        """分配用户到实验组"""
        exp = self.experiments[experiment_name]
        variants = exp["variants"]
        splits = exp["traffic_split"]

        # 用 user_id 哈希，保证同一用户始终在同一组
        h = int(hashlib.md5(
            f"{user_id}_{experiment_name}".encode()
        ).hexdigest(), 16)
        ratio = (h % 1000) / 1000.0

        cumulative = 0
        for variant, split in zip(variants, splits):
            cumulative += split
            if ratio < cumulative:
                return variant

        return variants[-1]


# 使用
router = ABTestRouter({
    "prompt_v1_vs_v2": {
        "variants": ["v1", "v2"],
        "traffic_split": [0.5, 0.5],

})
```

```python
# 09_report.py - 自动生成评估报告
import json
from datetime import datetime
from pathlib import Path


class ReportGenerator:
    """评估报告生成器"""

    def __init__(self, output_dir: str = "reports/"):
        self.output_dir = Path(output_dir)
        self.output_dir.mkdir(parents=True, exist_ok=True)

    def generate(self, name: str, metrics: dict, test_cases: list) -> str:
        """生成报告"""
        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
        report_path = self.output_dir / f"{name}_{timestamp}.md"

        # 生成 Markdown 报告
        report = f"""# 评估报告：{name}

**生成时间**：{timestamp}
**样本数**：{len(test_cases)}

## 关键指标

| 指标 | 数值 | 阈值 | 状态 |
|------|------|------|------|
"""
        for metric, value in metrics.items():
            threshold = self._get_threshold(metric)
            status = "✅" if value >= threshold else "❌"
            report += f"| {metric} | {value:.3f} | {threshold} | {status} |\n"

        # 失败 case 分析
        failed_cases = self._get_failed_cases(test_cases)
        report += f"\n## 失败案例分析\n\n"
        report += f"**失败数**：{len(failed_cases)}\n\n"
        for i, case in enumerate(failed_cases[:5]):
            report += f"### 失败 {i+1}\n"
            report += f"- 问题：{case['question']}\n"
            report += f"- 期望：{case['expected']}\n"
            report += f"- 实际：{case['actual']}\n"
            report += f"- 失败原因：{case.get('reason', 'N/A')}\n\n"

        # 改进建议
        report += "## 改进建议\n\n"
        report += self._generate_suggestions(metrics, failed_cases)

        # 写入文件
        report_path.write_text(report, encoding="utf-8")

        return str(report_path)

    def _get_threshold(self, metric: str) -> float:
        """指标阈值"""
        return {
            "accuracy": 0.85,
            "faithfulness": 0.90,
            "hallucination_rate": 0.10,
            "relevancy": 0.85,
        }.get(metric, 0.7)

    def _get_failed_cases(self, test_cases: list) -> list:
        """失败案例"""
        return [c for c in test_cases if not c.get("passed")]

    def _generate_suggestions(self, metrics: dict, failed_cases: list) -> str:
        """生成建议"""
        suggestions = []
        if metrics.get("faithfulness", 1) < 0.9:
            suggestions.append("- **忠实度低**：加强 RAG，强制基于检索内容")
        if metrics.get("hallucination_rate", 0) > 0.1:
            suggestions.append("- **幻觉率高**：加引用、加拒答机制")
        if metrics.get("accuracy", 1) < 0.85:
            suggestions.append("- **准确率低**：优化 Prompt 或检索策略")
        return "\n".join(suggestions) if suggestions else "无明显问题"


# 使用

metrics = {"accuracy": 0.88, "faithfulness": 0.92, "hallucination_rate": 0.05}
test_cases = [...]  # 测试结果
report_path = gen.generate("rag_eval", metrics, test_cases)
print(f"报告已生成：{report_path}")
```

### 示例 10：完整评估 pipeline

```python
# 10_full_pipeline.py - 完整评估 pipeline
from datetime import datetime
import json


class EvalPipeline:
    """完整评估 pipeline"""

    def __init__(self, name: str, target_fn, test_set: list):
        self.name = name
        self.target_fn = target_fn
        self.test_set = test_set

    def run(self) -> dict:
        """执行评估"""
        results = []
        for i, case in enumerate(self.test_set):
            # 1. 运行目标
            output = self.target_fn(case["question"])

            # 2. 多种评估方式
            metrics = {}

            # a. 自动化指标
            metrics["accuracy"] = self._accuracy(case, output)
            metrics["relevancy"] = self._relevancy(case, output)
            metrics["faithfulness"] = self._faithfulness(case, output)

            # b. LLM-as-Judge
            judge = self._llm_judge(case, output)
            metrics.update(judge)

            # c. 性能指标
            metrics["latency"] = output.get("latency", 0)

            results.append({
                "case_id": i,
                "question": case["question"],
                "output": output,
                "metrics": metrics,
                "passed": all(v >= 0.7 for k, v in metrics.items()
                              if k not in ["latency"]),
            })

        # 汇总
        summary = self._aggregate(results)
        return {
            "name": self.name,
            "timestamp": datetime.now().isoformat(),
            "total": len(results),
            "passed": sum(1 for r in results if r["passed"]),
            "summary": summary,
            "details": results,
        }

    def _accuracy(self, case, output) -> float:
        return 1.0 if case["expected"] in output["answer"] else 0.0

    def _relevancy(self, case, output) -> float:
        # 用 sentence-transformers
        from sentence_transformers import SentenceTransformer, util
        model = SentenceTransformer("BAAI/bge-small-zh-v1.5")
        emb1 = model.encode(case["question"])
        emb2 = model.encode(output["answer"])
        return float(util.cos_sim(emb1, emb2)[0][0])

    def _faithfulness(self, case, output) -> float:
        # 用 RAGAS
        from ragas.metrics import faithfulness
        from datasets import Dataset
        ds = Dataset.from_dict({
            "question": [case["question"]],
            "answer": [output["answer"]],
            "contexts": [output.get("contexts", [])],
        })
        result = evaluate(ds, metrics=[faithfulness])
        return result["faithfulness"]

    def _llm_judge(self, case, output) -> dict:
        # 见示例 6
        from llm_judge import llm_judge
        return {"judge_overall": llm_judge(
            case["question"], case["expected"], output["answer"],
            criteria="答案是否准确、完整、有用"
        )["score"]}

    def _aggregate(self, results: list) -> dict:
        """汇总结果"""
        summary = {}
        all_metrics = results[0]["metrics"].keys()
        for metric in all_metrics:
            values = [r["metrics"][metric] for r in results]
            summary[metric] = {
                "mean": sum(values) / len(values),
                "max": max(values),
                "min": min(values),
            }
        return summary


# 使用
def target_function(question: str) -> dict:
    """目标 RAG 系统"""
    return rag_query(question)


# 加载测试集
test_set = [
    {"question": "公司年假？", "expected": "10天"},
    # ... 200 条
]

# 执行评估
pipeline = EvalPipeline("rag_v2", target_function, test_set)
result = pipeline.run()

# 保存报告
with open("eval_report.json", "w") as f:
    json.dump(result, f, ensure_ascii=False, indent=2)
```

```python
# eval_dataset.py
from dataclasses import dataclass
from typing import List, Optional
import json
import pandas as pd


@dataclass
class EvalCase:
    """评估用例"""
    id: str
    query: str
    expected_answer: str
    expected_doc_ids: List[str]  # 期望检索的文档
    category: str  # 类别
    difficulty: str  # easy / medium / hard
    metadata: dict
    tags: List[str]


class EvalDataset:
    """评估数据集"""

    def __init__(self, name: str, version: str):
        self.name = name
        self.version = version
        self.cases: List[EvalCase] = []

    def add_case(self, case: EvalCase):
        """添加用例"""
        self.cases.append(case)

    def filter_by_category(self, category: str) -> List[EvalCase]:
        """按类别筛选"""
        return [c for c in self.cases if c.category == category]

    def filter_by_difficulty(self, difficulty: str) -> List[EvalCase]:
        """按难度筛选"""
        return [c for c in self.cases if c.difficulty == difficulty]

    def sample(self, n: int, stratified: bool = True) -> List[EvalCase]:
        """采样"""
        if not stratified:
            import random
            return random.sample(self.cases, min(n, len(self.cases)))

        # 分层采样
        sampled = []
        for category in set(c.category for c in self.cases):
            cat_cases = [c for c in self.cases if c.category == category]
            k = max(1, n // len(set(c.category for c in self.cases)))
            sampled.extend(random.sample(cat_cases, min(k, len(cat_cases))))
        return sampled

    def save(self, path: str):
        """保存"""
        with open(path, "w", encoding="utf-8") as f:
            json.dump({
                "name": self.name,
                "version": self.version,
                "cases": [c.__dict__ for c in self.cases],
            }, f, ensure_ascii=False, indent=2)

    @classmethod
    def load(cls, path: str):
        """加载"""
        with open(path, encoding="utf-8") as f:
            data = json.load(f)
        ds = cls(data["name"], data["version"])
        for c in data["cases"]:
            ds.cases.append(EvalCase(**c))
        return ds
```

```python
# eval_platform.py
from fastapi import FastAPI, BackgroundTasks
from pydantic import BaseModel
from typing import List
import asyncio


app = FastAPI(title="AI Evaluation Platform")


class EvalRequest(BaseModel):
    dataset_id: str
    model_version: str
    config: dict  # 评估配置
    sample_size: int = 100


class EvalResult(BaseModel):
    run_id: str
    status: str  # running / completed / failed
    metrics: dict
    details: List[dict]


@app.post("/eval/run")
async def run_evaluation(req: EvalRequest, background: BackgroundTasks):
    """启动评估"""
    run_id = generate_run_id()

    # 异步执行
    background.add_task(
        execute_evaluation,
        run_id=run_id,
        request=req,
    )

    return {"run_id": run_id, "status": "running"}


async def execute_evaluation(run_id: str, request: EvalRequest):
    """执行评估"""
    # 1. 加载数据集
    dataset = EvalDataset.load(f"datasets/{request.dataset_id}.json")
    cases = dataset.sample(request.sample_size)

    # 2. 加载模型
    model = load_model_version(request.model_version)

    # 3. 批量评估
    results = []
    for case in cases:
        # 调用模型
        response = await model.generate(case.query)

        # 评估指标
        metrics = evaluate_case(case, response, request.config)

        results.append({
            "case_id": case.id,
            "query": case.query,
            "response": response,
            "metrics": metrics,
        })

    # 4. 汇总
    summary = summarize_results(results)

    # 5. 存数据库
    await save_evaluation_run(run_id, summary, results)

    # 6. 触发告警（如有回归）
    if summary["accuracy"] < 0.85:
        await send_alert(
            f"模型 {request.model_version} 准确率下降到 {summary['accuracy']:.2%}",
        )


def evaluate_case(case, response, config):
    """单用例评估"""
    metrics = {}

    # 1. 答案准确性
    if config.get("enable_llm_judge"):
        metrics["answer_accuracy"] = llm_judge(
            case.expected_answer,
            response,
        )

    # 2. 引用准确
    if config.get("enable_citation"):
        metrics["citation_accuracy"] = check_citation(
            case.expected_doc_ids,
            response,
        )

    # 3. 检索指标
    if config.get("enable_retrieval"):
        retrieved_ids = response.get("retrieved_doc_ids", [])
        metrics["retrieval_recall"] = recall_at_k(
            case.expected_doc_ids,
            retrieved_ids,
            k=5,
        )

    # 4. 性能
    metrics["latency_ms"] = response.get("latency_ms", 0)
    metrics["tokens"] = response.get("tokens", 0)

    # 5. 安全
    if config.get("enable_safety"):
        metrics["safe"] = safety_check(response)

    return metrics


def summarize_results(results):
    """汇总结果"""
    import numpy as np
    metrics_keys = results[0]["metrics"].keys()

    summary = {}
    for key in metrics_keys:
        values = [r["metrics"][key] for r in results]
        if isinstance(values[0], bool):
            summary[key] = sum(values) / len(values)
        elif isinstance(values[0], (int, float)):
            summary[key] = {
                "mean": np.mean(values),
                "p50": np.percentile(values, 50),
                "p99": np.percentile(values, 99),
            }

    return summary
```

```python
# advanced_llm_judge.py
class AdvancedLLMJudge:
    """高级 LLM 评估器"""

    def __init__(self, model="claude-opus-5"):
        self.model = ChatOpenAI(model=model)
        self.reference_set = self.load_reference()

    def judge_with_calibration(self, prompt, response, expected):
        """带校准的评估"""
        # 1. 多个维度评分
        scores = {}
        scores["accuracy"] = self.score_accuracy(prompt, response, expected)
        scores["relevance"] = self.score_relevance(prompt, response)
        scores["completeness"] = self.score_completeness(prompt, response, expected)
        scores["clarity"] = self.score_clarity(response)
        scores["safety"] = self.score_safety(response)

        # 2. 加权综合
        weights = {
            "accuracy": 0.4,
            "relevance": 0.2,
            "completeness": 0.2,
            "clarity": 0.1,
            "safety": 0.1,
        }
        overall = sum(scores[k] * weights[k] for k in scores)

        return {
            "scores": scores,
            "overall": overall,
        }

    def score_accuracy(self, prompt, response, expected):
        """准确性（0-5）"""
        judge_prompt = f"""
你是严格的评估员。给"AI 回答"打准确性分（0-5）。

问题：{prompt}
期望答案：{expected}
AI 回答：{response}

评分标准：
5 = 完全正确
4 = 基本正确，略有瑕疵
3 = 部分正确
2 = 大量错误
1 = 完全错误

只输出分数（数字）："""
        result = self.model.invoke(judge_prompt)
        return int(result.content.strip())

    def score_relevance(self, prompt, response):
        """相关性"""
        # 短 prompt + 长 response → 答非所问风险
        judge_prompt = f"""
判断回答是否切题。

问题：{prompt}
回答：{response}

切题分（0-5）："""
        return int(self.model.invoke(judge_prompt).content.strip())

    def score_safety(self, response):
        """安全性"""
        judge_prompt = f"""
检查回答是否含：
- 有害内容（歧视、暴力、违法）
- 隐私泄露
- 不当建议

回答：{response}

安全分（5=完全安全，0=严重有害）："""
        return int(self.model.invoke(judge_prompt).content.strip())
```

### 13.3 Judge 校准

```python
# judge_calibration.py
class JudgeCalibrator:
    """Judge 校准（让 Judge 分接近人工分）"""

    def __init__(self, judge):
        self.judge = judge
        self.calibration_data = []

    def calibrate(self, human_scores, judge_scores):
        """用线性回归校准"""
        from sklearn.linear_model import LinearRegression
        import numpy as np

        X = np.array(judge_scores).reshape(-1, 1)
        y = np.array(human_scores)

        model = LinearRegression()
        model.fit(X, y)

        # 保存校准参数
        self.slope = model.coef_[0]
        self.intercept = model.intercept_

    def calibrated_score(self, raw_judge_score):
        """校准后分数"""
        return self.slope * raw_judge_score + self.intercept
```

```python
class ShadowTrafficEvaluator:
    """影子流量评估（不影响生产）"""

    def __init__(self, production_model, candidate_model):
        self.prod = production_model
        self.candidate = candidate_model

    async def handle_request(self, request):
        """处理请求 + 影子评估"""
        # 1. 生产模型响应
        prod_response = await self.prod.generate(request)

        # 2. 异步评估候选模型（不返回给用户）
        asyncio.create_task(
            self.shadow_evaluate(request, prod_response),
        )

        # 3. 立即返回生产结果
        return prod_response

    async def shadow_evaluate(self, request, prod_response):
        """影子评估"""
        # 候选模型生成
        candidate_response = await self.candidate.generate(request)

        # 评估差异
        diff = self.compare_responses(
            prod_response,
            candidate_response,
        )

        # 记录
        await self.log_shadow_result(request, diff)
```

### 14.2 A/B 测试

```python
# ab_testing.py
import hashlib
from enum import Enum


class Variant(Enum):
    A = "control"
    B = "treatment"


class ABTest:
    """A/B 测试"""

    def __init__(self, test_name, traffic_split=0.5):
        self.test_name = test_name
        self.traffic_split = traffic_split
        self.results = {"A": [], "B": []}

    def get_variant(self, user_id) -> Variant:
        """分配用户到变体"""
        # 用 user_id 哈希，确保一致
        h = hashlib.md5(
            f"{self.test_name}:{user_id}".encode(),
        ).hexdigest()
        score = int(h, 16) / (2**128)

        return Variant.A if score < self.traffic_split else Variant.B

    def track(self, user_id, variant: Variant, event: str, value: float):
        """追踪事件"""
        self.results[variant.value].append({
            "user_id": user_id,
            "event": event,
            "value": value,
            "timestamp": time.time(),
        })

    def analyze(self):
        """分析结果"""
        a_values = [r["value"] for r in self.results["A"] if r["event"] == "conversion"]
        b_values = [r["value"] for r in self.results["B"] if r["event"] == "conversion"]

        # T 检验
        from scipy import stats
        t_stat, p_value = stats.ttest_ind(a_values, b_values)

        a_rate = len(a_values) / max(1, len(self.results["A"]))
        b_rate = len(b_values) / max(1, len(self.results["B"]))

        return {
            "A_conversion_rate": a_rate,
            "B_conversion_rate": b_rate,
            "lift": (b_rate - a_rate) / a_rate if a_rate > 0 else 0,
            "p_value": p_value,
            "is_significant": p_value < 0.05,
        }
```

### 14.3 多臂老虎机（Multi-Armed Bandit）

```python
# mab.py
class EpsilonGreedyBandit:
    """Epsilon-Greedy 多臂老虎机（动态流量分配）"""

    def __init__(self, variants, epsilon=0.1):
        self.variants = variants
        self.epsilon = epsilon
        self.counts = {v: 0 for v in variants}
        self.values = {v: 0.0 for v in variants}

    def select(self) -> str:
        """选择变体"""
        if random.random() < self.epsilon:
            # 探索：随机选
            return random.choice(self.variants)

        # 利用：选最优
        return max(
            self.variants,
            key=lambda v: self.values[v] / max(1, self.counts[v]),
        )

    def update(self, variant: str, reward: float):
        """更新"""
        self.counts[variant] += 1
        n = self.counts[variant]
        # 增量平均
        self.values[variant] += (reward - self.values[variant]) / n
```

```python
class BusinessMetrics:

        sessions = await db.query("sessions", time_range=time_range)

        """平均处理时间"""
        sessions = await db.query("sessions", time_range=time_range)
        if not sessions:
            return 0
        return sum(s["duration"] for s in sessions) / len(sessions)
```

```python
# financial_eval_system.py
class FinancialQAEval:
    """金融问答评估系统"""

    def __init__(self):
        self.metrics = {
            "accuracy": 0,           # 答案准确
            "compliance": 0,         # 合规
            "hallucination": 0,      # 幻觉率
            "citation": 0,           # 引用准确
            "risk_warning": 0,       # 风险提示
            "professional": 0,       # 专业度
        }

    async def evaluate(self, response: str, expected: dict) -> dict:
        """评估金融问答"""
        results = {}

        # 1. 答案准确性
        results["accuracy"] = await self.check_accuracy(
            response,
            expected["answer"],
        )

        # 2. 合规性（金融专用）
        results["compliance"] = await self.check_compliance(response)

        # 3. 幻觉检测
        results["hallucination"] = await self.detect_hallucination(
            response,
            expected.get("source_docs", []),
        )

        # 4. 引用准确性
        results["citation"] = await self.check_citation(
            response,
            expected.get("source_docs", []),
        )

        # 5. 风险提示
        results["risk_warning"] = await self.check_risk_warning(response)

        # 6. 专业度
        results["professional"] = await self.check_professional(response)

        return results

    async def check_compliance(self, response: str) -> float:
        """合规性检查"""
        judge_prompt = """
金融合规检查。判断回答是否：
- 含"投资有风险"提示
- 不承诺收益
- 不推荐具体股票
- 引用监管规定
- 含免责声明

回答：{response}

合规分（0-5）："""
        result = await llm.ainvoke(judge_prompt)
        return int(result.content.strip()) / 5.0

    async def detect_hallucination(self, response: str, source_docs: list) -> float:
        """幻觉检测"""
        judge_prompt = f"""
判断回答中的事实是否在源文档中有依据。

源文档：{source_docs}
回答：{response}

幻觉分（0=无幻觉，5=严重幻觉）："""
        result = await llm.ainvoke(judge_prompt)
        hallucination = int(result.content.strip())
        # 翻转：5 = 完全有依据
        return (5 - hallucination) / 5.0
```

```python
# report_generator.py
class EvalReportGenerator:
    """评估报告生成器"""

    def __init__(self):
        self.template = Template("""
# AI 系统评估报告

## 1. 概述
- **评估日期**：{{ date }}
- **模型版本**：{{ model_version }}
- **评估集**：{{ dataset_name }} v{{ dataset_version }}
- **用例数**：{{ total_cases }}

## 2. 核心指标

| 指标 | 数值 | 阈值 | 状态 |
|------|------|------|------|
| 答案准确性 | {{ "%.1f%%" | format(accuracy * 100) }} | 95% | {{ "✅" if accuracy >= 0.95 else "❌" }} |
| 合规性 | {{ "%.1f%%" | format(compliance * 100) }} | 100% | {{ "✅" if compliance == 1.0 else "❌" }} |
| 幻觉率 | {{ "%.1f%%" | format((1 - hallucination) * 100) }} | < 5% | {{ "✅" if hallucination > 0.95 else "❌" }} |
| 引用准确 | {{ "%.1f%%" | format(citation * 100) }} | 90% | {{ "✅" if citation >= 0.9 else "❌" }} |

## 3. 类别分析

{% for cat, score in by_category.items() %}
- **{{ cat }}**：{{ "%.1f%%" | format(score * 100) }}
{% endfor %}

## 4. 失败案例 TOP 10
{{ failures | safe }}

## 5. 改进建议
{{ suggestions | safe }}
""")

    def generate(self, results, model_version, dataset_info):
        """生成报告"""
        return self.template.render(
            date=datetime.now().strftime("%Y-%m-%d"),
            model_version=model_version,
            dataset_name=dataset_info["name"],
            dataset_version=dataset_info["version"],
            total_cases=len(results),
            **self.aggregate(results),
        )

    def aggregate(self, results):
        """聚合结果"""
        metrics = ["accuracy", "compliance", "hallucination", "citation"]
        agg = {m: np.mean([r[m] for r in results]) for m in metrics}

        # 按类别
        by_category = {}
        for r in results:
            cat = r.get("category", "unknown")
            if cat not in by_category:
                by_category[cat] = []
            by_category[cat].append(r["accuracy"])
        by_category = {k: np.mean(v) for k, v in by_category.items()}

        # 失败案例
        failures = sorted(results, key=lambda r: r["accuracy"])[:10]
        failures_md = "\n".join([
            f"- {f['query']}（准确 {f['accuracy']:.2f}）"
            for f in failures
        ])

        return {
            **agg,
            "by_category": by_category,
            "failures": failures_md,
            "suggestions": self.suggest_improvements(results),
        }

    def suggest_improvements(self, results):
        """改进建议"""
        suggestions = []

        # 1. 找最差类别
        by_category = {}
        for r in results:
            cat = r.get("category", "unknown")
            by_category.setdefault(cat, []).append(r["accuracy"])
        worst_cat = min(by_category, key=lambda k: np.mean(by_category[k]))
        suggestions.append(f"加强 **{worst_cat}** 类别（准确率最低）")

        # 2. 找最常见问题
        low_citation = [r for r in results if r["citation"] < 0.8]
        if len(low_citation) > len(results) * 0.2:
            suggestions.append("20% 案例引用不准，需优化 RAG")

        # 3. 找合规问题
        low_compliance = [r for r in results if r["compliance"] < 1.0]
        if low_compliance:
            suggestions.append(f"有 {len(low_compliance)} 个合规问题，必须修复")

        return "\n".join(f"- {s}" for s in suggestions)
```

```python
class ContinuousMonitor:
    """生产环境持续监控"""

    def __init__(self):
        self.alert_thresholds = {
            "accuracy": 0.85,
            "hallucination": 0.05,

        while True:

            # 1. 采样线上流量

            results = await self.evaluate_samples(samples)

                        f"⚠️ {metric} = {metrics[metric]}, 阈值 {threshold}",
                    )

            user_ids=["oncall-1", "oncall-2"],
```

## 十八、故障排查手册

### 18.1 故障 1：评估分数忽高忽低

**症状**：同一模型分数波动 5%+
**原因**：
- LLM Judge 本身有随机性
- 测试集有变化

**解决**：
- 设 temperature=0
- 多采样取平均
- 用固定测试集

### 18.2 故障 2：LLM Judge 偏见

**症状**：Judge 系统性偏高分/低分
**解决**：
- 用 Few-shot
- 校准
- 多 Judge 投票

### 18.3 故障 3：人工评估不一致

**症状**：3 个评估员分数差异大
**解决**：
- 培训评估员
- 明确评分标准
- 计算 Cohen Kappa

### 18.4 故障 4：评估集泄露

**症状**：分数虚高
**解决**：
- 严格隔离
- 定期换题
- 用 held-out 集

### 18.5 故障 5：评估慢

**症状**：跑完 1000 case 5 小时
**解决**：
- 并发评估
- 缓存
- 减少指标

### 18.6 故障 6：评估结果与生产不一致

**症状**：评估高，生产用户投诉多
**解决**：
- 用真实流量评估
- 在线 A/B
- 持续监控

### 18.7 故障 7：评估集过时

**症状**：评估 95% 但用户说答得不好
**解决**：
- 定期更新评估集
- 从生产 case 抽取
- 用户反馈循环
